# ── Coffee Shop Agentic RAG System ──────────────────────────────
from langgraph.graph import StateGraph, END
from langchain_pinecone import PineconeVectorStore
from fastapi import FastAPI
from supabase import create_client

# Agent state with dual-memory architecture
class AgentState(TypedDict):
    messages:       list[BaseMessage]
    short_term:     list        # In-session conversation history
    user_profile:   dict        # Persistent Supabase long-term memory
    retrieved_docs: list        # Pinecone RAG context
    agent_type:     str         # Router decision output

# Build LangGraph state machine
graph = StateGraph(AgentState)
graph.add_node("router",        router_agent)
graph.add_node("order_agent",   order_processing_agent)
graph.add_node("menu_agent",    menu_query_agent)
graph.add_node("rag_retriever", agentic_rag_node)
graph.add_conditional_edges("router", route_decision)

app = FastAPI()
pipeline = graph.compile()
print("✓ System compiled | Agents: 3 | Memory: dual | Status: deployed")
